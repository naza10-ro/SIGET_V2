class AdminMiddleware
{
    /**
     * @param  Request  $request
     * @param  Closure  $next
     * @param  string   ...$roles  Roles permitidos para esta ruta.
     *                             Se pasan como: 'role:admin,operador'
     */
    public function handle(Request $request, Closure $next, string ...$roles): Response
    {
        $user = $request->user();

        // VerificaciÃ³n 1: usuario autenticado
        // (Sanctum ya garantiza esto con auth:sanctum, pero es buena prÃ¡ctica)
        if (! $user) {
            return response()->json([
                'message' => __('auth.unauthenticated'),
            ], 401);
        }

        // VerificaciÃ³n 2: el rol del usuario estÃ¡ entre los permitidos
        if (! in_array($user->rol, $roles, strict: true)) {
            return response()->json([
                'message' => __('auth.unauthorized'),
                'errors'  => [
                    'rol' => [
                        __('auth.unauthorized_role', [
                            'rol'      => $user->rol,
                            'required' => implode(' o ', $roles),
                        ]),
                    ],
                ],
            ], 403);
        }

        return $next($request);
    }

     *

     *  - Encuesta de gas: critico (hay gas vencido) Â· en_progreso (hay por
     *    vencer) Â· pendiente (solo falta historial para estimar).
     *  - Otras encuestas: critico (â‰¥ 50 % de familias sin entrega) Â·

    public function tareas(Request $request, EstimadorConsumoService $estimador): JsonResponse

        $familiaIds = Familia::whereIn('sede_id', $sedes)->pluck('id');
        $total      = $familiaIds->count();

        if ($total === 0) {
            return response()->json(['data' => []], 200);
        }

        foreach (Encuesta::activas()->orderBy('fecha_cierre')->orderBy('id')->get() as $encuesta) {
            if ($encuesta->esDeGas()) {
                $conteo   = $estimador->estimarPorFamilias($familiaIds, $encuesta->tipo)->countBy('estado');
                $vencidas = (int) ($conteo['vencido'] ?? 0);
                $proximas = (int) ($conteo['proximo'] ?? 0);
                $sinDatos = (int) ($conteo['sin_datos'] ?? 0);

                if ($vencidas + $proximas + $sinDatos === 0) {
                    continue; // todas al dÃ­a
                }

                $estado = match (true) {
                    $vencidas > 0 => 'critico',
                    $proximas > 0 => 'en_progreso',
                    default       => 'pendiente',
                };

                $descripcion = "{$vencidas} familia(s) con gas vencido, {$proximas} por vencer y {$sinDatos} sin historial suficiente (de {$total}).";

                    ->whereIn('familia_id', $familiaIds)
                    ->distinct()

                    continue; // cobertura completa
                }

                $estado      = ($pendientes / $total) >= 0.5 ? 'critico' : 'en_progreso';
                $descripcion = "{$pendientes} de {$total} familias todavÃ­a sin entrega registrada.";
            }

                'titulo'       => $encuesta->nombre,
                'descripcion'  => $descripcion,
                'estado'       => $estado,
                'fecha_limite' => $encuesta->fecha_cierre?->toDateString(),
            ];
        }

        return response()->json(['data' => $tareas], 200);
    }

    private function sedesDelVocero(Request $request): array
    {
        $habitanteId = $request->user()->habitante_id;

        $sedes = $habitanteId
            ? Vocero::activos()->where('habitante_id', $habitanteId)->pluck('sede_id')->unique()->values()->all()
            : [];

        if ($sedes === []) {
            abort(403, 'No se encontrÃ³ un registro de vocero activo asociado a este usuario.');
        }

        return array_map('intval', $sedes);
    }

        $dias = $fechas
            ->map(fn ($f) => Carbon::parse($f)->startOfDay())
            ->unique(fn (Carbon $d) => $d->toDateString())
            ->sortBy(fn (Carbon $d) => $d->getTimestamp())
            ->values();

            return [
                'familia_id'                 => $familiaId,
                'estado'                     => 'sin_datos',
                'promedio_dias_duracion'     => null,
                'ultima_entrega'             => $dias->last()?->toDateString(),
                'fecha_estimada_agotamiento' => null,
                'dias_restantes'             => null,
            ];
        }

        $sumaDias = 0;
        for ($i = 1; $i < $dias->count(); $i++) {
            $sumaDias += (int) round(
                ($dias[$i]->getTimestamp() - $dias[$i - 1]->getTimestamp()) / self::SEGUNDOS_POR_DIA
            );
        }

        $promedio      = max(1, (int) round($sumaDias / ($dias->count() - 1)));
        $ultima        = $dias->last();
        $fechaEstimada = $ultima->copy()->addDays($promedio);
        $diasRestantes = (int) round(
            ($fechaEstimada->getTimestamp() - Carbon::today()->getTimestamp()) / self::SEGUNDOS_POR_DIA
        );

            'familia_id'                 => $familiaId,
            'estado'                     => $this->clasificarEstado($diasRestantes),
            'promedio_dias_duracion'     => $promedio,
            'ultima_entrega'             => $ultima->toDateString(),
            'fecha_estimada_agotamiento' => $fechaEstimada->toDateString(),
            'dias_restantes'             => $diasRestantes,
        ];

    private function riesgoDesdeConteo(array $conteo, int $total): ?int
    {

        }

        return match (true) {
            $conteo['vencido'] > 0 => 2,
            $conteo['proximo'] > 0 => 1,
            default                => 0,
        };
    }
